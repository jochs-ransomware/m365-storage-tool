const ALLOWED_ORIGIN = 'https://yellow-glacier-0b7f79b10.6.azurestaticapps.net'; const GRAPH = 'https://graph.microsoft.com/v1.0'; const PERIOD = 'D30'; export default { async fetch(request, env, ctx) { if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': ALLOWED_ORIGIN, 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', } });
